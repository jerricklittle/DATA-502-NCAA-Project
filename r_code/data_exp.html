Plot 5: Percent Difference in Expense Categories from NWC Median

  • Geometries: Columns, vline
  • Aesthetics: x = difference in percentage, y = expense category, fill = positive/negative
expense_breakdown = d3_instLevel %>%
  filter(institution_name %in% nwc | institution_name == "Willamette University") %>%
  mutate(
    hc_salaries =
      coalesce(HDCOACH_SALARY_MEN * NUM_HDCOACH_MEN, 0) + #Making 0 if NA
      coalesce(HDCOACH_SALARY_WOMEN * NUM_HDCOACH_WOMEN, 0) +
      coalesce(HDCOACH_SALARY_COED * NUM_HDCOACH_COED, 0),

    ac_salaries =
      coalesce(ASCOACH_SALARY_MEN * NUM_ASCOACH_MEN, 0) +
      coalesce(ASCOACH_SALARY_WOMEN * NUM_ASCOACH_WOMEN, 0) +
      coalesce(ASCOACH_SALARY_COED * NUM_ASCOACH_COED, 0),

    recruiting =
      coalesce(RECRUITEXP_MEN, 0) +
      coalesce(RECRUITEXP_WOMEN, 0) +
      coalesce(RECRUITEXP_COED, 0),

    operating = IL_TOTAL_OPEXP_INCLCOED,
    
    other = GRND_TOTAL_EXPENSE - operating - recruiting - hc_salaries - ac_salaries
  )


selected_year = 2025 #User chooses
title = paste0("Where Willamette's Athletic Spending Differed in ", selected_year) #User chooses

expense_breakdown %>%
  filter(year == selected_year) %>%
  mutate(
    operating = operating / GRND_TOTAL_EXPENSE * 100,
    recruiting = recruiting / GRND_TOTAL_EXPENSE * 100,
    hc_salaries = hc_salaries / GRND_TOTAL_EXPENSE * 100,
    ac_salaries = ac_salaries / GRND_TOTAL_EXPENSE * 100,
    other = other / GRND_TOTAL_EXPENSE * 100
  ) %>%
  select(
    institution_name,
    operating,
    recruiting,
    hc_salaries,
    ac_salaries,
    other
  ) %>%
  pivot_longer(
    cols = -institution_name,
    names_to = "category",
    values_to = "percentage"
  ) %>%
  group_by(category) %>%
  mutate(
    nwc_median = median(
      percentage[institution_name %in% nwc],
      na.rm = TRUE
    )
  ) %>%
  ungroup() %>%
  filter(institution_name == "Willamette University") %>%
  mutate(
    difference = percentage - nwc_median,
    category = recode(
      category,
      operating = "Operating expenses",
      recruiting = "Recruiting",
      hc_salaries = "Head coach salaries",
      ac_salaries = "Assistant coach salaries",
      other = "Other expenses"
    )
  ) %>%
  ggplot(aes(x = difference, y = reorder(category, difference), fill = difference > 0)) +
  geom_col() +
  geom_vline(xintercept = 0, linetype = "dashed") +
  scale_x_continuous(
    labels = \(x) paste0(x, " pp")
  ) +
  theme_light() +
  theme(
    legend.position = "none",
    plot.caption.position = "plot" #Moving caption to the right side
  ) +
  labs(
    title = "Coaching Salaries Has More Share in Willamette's Budget",
    subtitle = "Compared to NWC median",
    x = "Difference in percentage points",
    y = "Expense Category",
    caption = "Data retrieved from the Equity in Athletics Data Analysis Cutting Tool\nby the U.S. Department of Education (https://ope.ed.gov/athletics/)."
  )

Analysis While Willamette coaches need to be more invested in like our graph states above, the gap between expenses on athletes needs to be closed. As a result, more immediate change can be forecasted by increasing the percentage of our budget we spend on recruiting, operating expenses, and other expenses.
